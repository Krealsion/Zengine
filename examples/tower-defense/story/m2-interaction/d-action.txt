    void on(const ws::PaneActionRequested& asked, loom::Mail& mail) {
        if (!mail.authored_from_role(kWorkshop) || asked.pane != kPane) {
            return;
        }
        if (asked.id == "td.left") {
            move_cursor(state_, -1, 0);
        } else if (asked.id == "td.right") {
            move_cursor(state_, 1, 0);
        } else if (asked.id == "td.up") {
            move_cursor(state_, 0, -1);
        } else if (asked.id == "td.down") {
            move_cursor(state_, 0, 1);
        } else if (asked.id == "td.build") {
            build_at_cursor(state_);
        } else if (asked.id == "td.restart") {
            new_game(state_);
        }
        show(mail);

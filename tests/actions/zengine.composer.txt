# The action census's rows for zengine.composer; the form is tests/actions.txt's. ASCII only.
zengine.composer - compose.enter return - - choose
zengine.composer - compose.submit ctrl+return~ctrl+e - - submit
zengine.composer - compose.store ctrl+s InventoryAdd/1 - store command
zengine.composer - compose.preset ctrl+b InventoryAdd/1 - store preset (may be incomplete)
zengine.composer - compose.unset ctrl+u - - unset field and discard value
zengine.composer - compose.back escape - - back to the catalog
zengine.composer - compose.up up - - row up
zengine.composer - compose.down down - - row down
zengine.composer - compose.include tab - - include or leave out the field
